```
给你两棵二叉树的根节点 p 和 q ，编写一个函数来检验这两棵树是否相同。

如果两个树在结构上相同，并且节点具有相同的值，则认为它们是相同的。
```

```go
package main

import "fmt"

// TreeNode 定义二叉树节点
type TreeNode struct {
    Val   int
    Left  *TreeNode
    Right *TreeNode
}

// isSameTree 检查两棵二叉树是否相同
func isSameTree(p *TreeNode, q *TreeNode) bool {
    // 如果两个节点都是 nil，说明当前子树相同
    if p == nil && q == nil {
        return true
    }
    // 如果一个节点是 nil，另一个不是，说明结构不同
    if p == nil || q == nil {
        return false
    }
    // 如果当前节点值不同，说明树不同
    if p.Val != q.Val {
        return false
    }
    // 递归检查左子树和右子树
    return isSameTree(p.Left, q.Left) && isSameTree(p.Right, q.Right)
}

func main() {
    // 示例树
    tree1 := &TreeNode{Val: 1, Left: &TreeNode{Val: 2}, Right: &TreeNode{Val: 3}}
    tree2 := &TreeNode{Val: 1, Left: &TreeNode{Val: 2}, Right: &TreeNode{Val: 3}}

    fmt.Println(isSameTree(tree1, tree2)) // 输出: true
}
```

```js
// 定义二叉树节点
function TreeNode(val, left = null, right = null) {
    this.val = val;
    this.left = left;
    this.right = right;
}

// 检查两棵二叉树是否相同
function isSameTree(p, q) {
    // 如果两个节点都是 null，说明当前子树相同
    if (p === null && q === null) {
        return true;
    }
    // 如果一个节点是 null，另一个不是，说明结构不同
    if (p === null || q === null) {
        return false;
    }
    // 如果当前节点值不同，说明树不同
    if (p.val !== q.val) {
        return false;
    }
    // 递归检查左子树和右子树
    return isSameTree(p.left, q.left) && isSameTree(p.right, q.right);
}

// 示例树
const tree1 = new TreeNode(1, new TreeNode(2), new TreeNode(3));
const tree2 = new TreeNode(1, new TreeNode(2), new TreeNode(3));

console.log(isSameTree(tree1, tree2)); // 输出: true
```

```rust
// 定义二叉树节点
#[derive(PartialEq, Eq)]
struct TreeNode {
    val: i32,
    left: Option<Box<TreeNode>>,
    right: Option<Box<TreeNode>>,
}

impl TreeNode {
    fn new(val: i32) -> Self {
        TreeNode { val, left: None, right: None }
    }
}

// 检查两棵二叉树是否相同
fn is_same_tree(p: Option<&Box<TreeNode>>, q: Option<&Box<TreeNode>>) -> bool {
    match (p, q) {
        (None, None) => true,
        (Some(p_node), Some(q_node)) => {
            p_node.val == q_node.val
                && is_same_tree(p_node.left.as_ref(), q_node.left.as_ref())
                && is_same_tree(p_node.right.as_ref(), q_node.right.as_ref())
        }
        _ => false,
    }
}

fn main() {
    // 示例树
    let tree1 = Some(Box::new(TreeNode {
        val: 1,
        left: Some(Box::new(TreeNode::new(2))),
        right: Some(Box::new(TreeNode::new(3))),
    }));

    let tree2 = Some(Box::new(TreeNode {
        val: 1,
        left: Some(Box::new(TreeNode::new(2))),
        right: Some(Box::new(TreeNode::new(3))),
    }));

    println!("{}", is_same_tree(tree1.as_ref(), tree2.as_ref())); // 输出: true
}
```
