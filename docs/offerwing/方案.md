### 1. 系统级截图技术

**Windows平台实现：**
```python
import win32gui
import win32ui
import win32con
from PIL import Image
import numpy as np

class ScreenCapture:
    def __init__(self):
        self.hwnd = win32gui.GetDesktopWindow()
    
    def capture_screen_silent(self):
        """无感截取整个屏幕"""
        # 获取屏幕尺寸
        left, top, right, bottom = win32gui.GetWindowRect(self.hwnd)
        width = right - left
        height = bottom - top
        
        # 创建设备上下文
        hwndDC = win32gui.GetWindowDC(self.hwnd)
        mfcDC = win32ui.CreateDCFromHandle(hwndDC)
        saveDC = mfcDC.CreateCompatibleDC()
        
        # 创建位图对象
        saveBitMap = win32ui.CreateBitmap()
        saveBitMap.CreateCompatibleBitmap(mfcDC, width, height)
        saveDC.SelectObject(saveBitMap)
        
        # 截图到内存（关键：使用SRCCOPY模式，无闪烁）
        saveDC.BitBlt((0, 0), (width, height), mfcDC, (0, 0), win32con.SRCCOPY)
        
        # 转换为PIL图像
        bmpinfo = saveBitMap.GetInfo()
        bmpstr = saveBitMap.GetBitmapBits(True)
        img = Image.frombuffer('RGB', (bmpinfo['bmWidth'], bmpinfo['bmHeight']), 
                              bmpstr, 'raw', 'BGRX', 0, 1)
        
        # 清理资源
        win32gui.DeleteObject(saveBitMap.GetHandle())
        saveDC.DeleteDC()
        mfcDC.DeleteDC()
        win32gui.ReleaseDC(self.hwnd, hwndDC)
        
        return img
```

**跨平台实现（推荐）：**
```python
import mss
import numpy as np
from PIL import Image

class UniversalScreenCapture:
    def __init__(self):
        self.sct = mss.mss()
    
    def capture_silent(self, monitor_num=1):
        """
        无感截图 - 使用MSS库实现跨平台
        monitor_num: 显示器编号，1为主显示器
        """
        with self.sct:
            # 获取指定显示器信息
            monitor = self.sct.monitors[monitor_num]
            
            # 静默截图（内存操作，无UI闪烁）
            screenshot = self.sct.grab(monitor)
            
            # 转换为PIL图像
            img = Image.frombytes('RGB', screenshot.size, screenshot.bgra, 'raw', 'BGRX')
            return img
    
    def capture_region_silent(self, x, y, width, height):
        """截取指定区域"""
        region = {'top': y, 'left': x, 'width': width, 'height': height}
        screenshot = self.sct.grab(region)
        return Image.frombytes('RGB', screenshot.size, screenshot.bgra, 'raw', 'BGRX')
```

### 2. 智能区域识别

```python
import cv2
import numpy as np

class SmartRegionDetector:
    def __init__(self):
        self.last_screenshot = None
        self.question_regions = []
    
    def detect_question_areas(self, screenshot):
        """智能检测题目区域"""
        gray = cv2.cvtColor(np.array(screenshot), cv2.COLOR_RGB2GRAY)
        
        # 使用边缘检测找到文本区域
        edges = cv2.Canny(gray, 50, 150)
        
        # 形态学操作连接文本
        kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (20, 1))
        dilated = cv2.dilate(edges, kernel, iterations=1)
        
        # 找到轮廓
        contours, _ = cv2.findContours(dilated, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
        
        question_boxes = []
        for contour in contours:
            x, y, w, h = cv2.boundingRect(contour)
            # 过滤太小的区域
            if w > 100 and h > 20:
                question_boxes.append((x, y, w, h))
        
        return question_boxes
    
    def detect_changes(self, current_screenshot):
        """检测屏幕变化，只截取变化区域"""
        if self.last_screenshot is None:
            self.last_screenshot = current_screenshot
            return True, None
        
        # 计算差异
        diff = cv2.absdiff(
            np.array(self.last_screenshot), 
            np.array(current_screenshot)
        )
        
        # 如果差异超过阈值，说明内容发生变化
        change_ratio = np.sum(diff > 30) / diff.size
        
        if change_ratio > 0.01:  # 1%的像素发生变化
            self.last_screenshot = current_screenshot
            return True, diff
        
        return False, None
```

### 3. 后台服务实现

```python
import threading
import time
from queue import Queue

class BackgroundScreenMonitor:
    def __init__(self):
        self.capture = UniversalScreenCapture()
        self.detector = SmartRegionDetector()
        self.screenshot_queue = Queue()
        self.is_monitoring = False
        self.monitor_thread = None
    
    def start_monitoring(self, interval=0.5):
        """启动后台监控"""
        self.is_monitoring = True
        self.monitor_thread = threading.Thread(target=self._monitor_loop, args=(interval,))
        self.monitor_thread.daemon = True
        self.monitor_thread.start()
    
    def _monitor_loop(self, interval):
        """后台监控循环"""
        while self.is_monitoring:
            try:
                # 无感截图
                screenshot = self.capture.capture_silent()
                
                # 检测变化
                has_change, diff = self.detector.detect_changes(screenshot)
                
                if has_change:
                    # 检测题目区域
                    question_areas = self.detector.detect_question_areas(screenshot)
                    
                    # 将结果放入队列
                    self.screenshot_queue.put({
                        'timestamp': time.time(),
                        'screenshot': screenshot,
                        'question_areas': question_areas,
                        'has_change': has_change
                    })
                
                time.sleep(interval)
                
            except Exception as e:
                print(f"监控异常: {e}")
                time.sleep(1)
    
    def get_latest_screenshot(self):
        """获取最新截图"""
        if not self.screenshot_queue.empty():
            return self.screenshot_queue.get()
        return None
    
    def stop_monitoring(self):
        """停止监控"""
        self.is_monitoring = False
        if self.monitor_thread:
            self.monitor_thread.join()
```

### 5. 无感截图的优化技术

```python
class OptimizedCapture:
    def __init__(self):
        self.capture = UniversalScreenCapture()
        self.cache = {}
        self.last_hash = None
    
    def smart_capture(self):
        """智能截图 - 避免重复截图"""
        screenshot = self.capture.capture_silent()
        
        # 计算图像哈希
        img_hash = self.calculate_image_hash(screenshot)
        
        # 如果与上次截图相同，返回缓存
        if img_hash == self.last_hash:
            return self.cache.get('last_screenshot')
        
        # 更新缓存
        self.last_hash = img_hash
        self.cache['last_screenshot'] = screenshot
        
        return screenshot
    
    def calculate_image_hash(self, image):
        """计算图像哈希值"""
        import hashlib
        return hashlib.md5(image.tobytes()).hexdigest()
    
    def capture_with_compression(self):
        """压缩截图以提高传输效率"""
        screenshot = self.capture.capture_silent()
        
        # 压缩图像
        screenshot.thumbnail((1920, 1080), Image.Resampling.LANCZOS)
        
        return screenshot
```

### 4. **AI 集成优化**
```swift
import CoreML
import Vision

@MainActor
class AIAnswerEngine: ObservableObject {
    @Published var currentAnswer = ""
    @Published var detectedRegions: [CGRect] = []
    
    private var ocrRequest: VNRecognizeTextRequest?
    
    init() {
        setupOCR()
    }
    
    private func setupOCR() {
        ocrRequest = VNRecognizeTextRequest { [weak self] request, error in
            guard let observations = request.results as? [VNRecognizedTextObservation] else { return }
            
            Task { @MainActor in
                await self?.processOCRResults(observations)
            }
        }
        
        // 配置 OCR 参数
        ocrRequest?.recognitionLevel = .accurate
        ocrRequest?.recognitionLanguages = ["zh-Hans", "en-US"]
        ocrRequest?.usesLanguageCorrection = true
    }
    
    func analyzeImage(_ cgImage: CGImage) async {
        let handler = VNImageRequestHandler(cgImage: cgImage)
        
        do {
            try handler.perform([ocrRequest!])
        } catch {
            print("OCR 失败: \(error)")
        }
    }
    
    private func processOCRResults(_ observations: [VNRecognizedTextObservation]) async {
        var detectedText = ""
        var regions: [CGRect] = []
        
        for observation in observations {
            guard let topCandidate = observation.topCandidates(1).first else { continue }
            
            detectedText += topCandidate.string + "\n"
            regions.append(observation.boundingBox)
        }
        
        self.detectedRegions = regions
        
        if !detectedText.isEmpty {
            await generateAnswer(for: detectedText)
        }
    }
    
    private func generateAnswer(for question: String) async {
        // 调用 AI API 生成答案
        do {
            let answer = try await callAIAPI(question: question)
            self.currentAnswer = answer
        } catch {
            self.currentAnswer = "生成答案失败: \(error.localizedDescription)"
        }
    }
    
    private func callAIAPI(question: String) async throws -> String {
        // 实现 AI API 调用
        let url = URL(string: "https://api.openai.com/v1/chat/completions")!
        var request = URLRequest(url: url)
        request.httpMethod = "POST"
        request.setValue("Bearer YOUR_API_KEY", forHTTPHeaderField: "Authorization")
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        
        let payload = [
            "model": "gpt-4",
            "messages": [
                ["role": "user", "content": "请回答这道题：\(question)"]
            ]
        ]
        
        request.httpBody = try JSONSerialization.data(withJSONObject: payload)
```

```swift
        // 发送到语音识别引擎
        Task { @MainActor in
            await self.sendToSpeechRecognition(audioData: audioData)
        }
    }
```

```swift
class CoreAudioCapture {
    private var audioUnit: AudioUnit?

        // 1. 创建 Audio Unit
        var componentDescription = AudioComponentDescription()
        componentDescription.componentType = kAudioUnitType_Output
        componentDescription.componentSubType = kAudioUnitSubType_HALOutput
        componentDescription.componentManufacturer = kAudioUnitManufacturer_Apple
        
        guard let component = AudioComponentFindNext(nil, &componentDescription) else {
            throw AudioCaptureError.componentNotFound
        }
        
        var audioUnit: AudioUnit?
        try AudioComponentInstanceNew(component, &audioUnit).checkError()
        self.audioUnit = audioUnit
        
        // 2. 配置输入输出
        var enableInput: UInt32 = 1
        try AudioUnitSetProperty(
            audioUnit!,
            kAudioOutputUnitProperty_EnableIO,
            kAudioUnitScope_Input,
            1, // input bus
            &enableInput,
            UInt32(MemoryLayout<UInt32>.size)
        ).checkError()
        
        var disableOutput: UInt32 = 0
        try AudioUnitSetProperty(
            audioUnit!,
            kAudioOutputUnitProperty_EnableIO,
            kAudioUnitScope_Output,
            0, // output bus
            &disableOutput,
            UInt32(MemoryLayout<UInt32>.size)
        ).checkError()
        
        // 3. 设置音频格式
        setupAudioFormat()
        
        // 4. 设置回调函数
        var inputCallback = AURenderCallbackStruct(
            inputProc: audioInputCallback,
            inputProcRefCon: Unmanaged.passUnretained(self).toOpaque()
        )
        
        try AudioUnitSetProperty(
            audioUnit!,
            kAudioOutputUnitProperty_SetInputCallback,
            kAudioUnitScope_Global,
            0,
            &inputCallback,
            UInt32(MemoryLayout<AURenderCallbackStruct>.size)
        ).checkError()
    }
    
    private func setupAudioFormat() {
        audioFormat.mSampleRate = 44100.0
        audioFormat.mFormatID = kAudioFormatLinearPCM
        audioFormat.mFormatFlags = kAudioFormatFlagIsFloat | kAudioFormatFlagIsPacked
        audioFormat.mChannelsPerFrame = 2
        audioFormat.mBitsPerChannel = 32
        audioFormat.mBytesPerFrame = audioFormat.mChannelsPerFrame * 4
        audioFormat.mFramesPerPacket = 1
        audioFormat.mBytesPerPacket = audioFormat.mBytesPerFrame
    }
}
```

```swift
class DualAudioProcessor: ObservableObject {
    private var systemAudioCapture = SystemAudioCapture()
    private var microphoneCapture = MicrophoneCapture()
    private var audioMixer = AudioMixer()
    
    @Published var interviewerAudio: [Float] = []
    @Published var candidateAudio: [Float] = []

        // 1. 同时启动系统音频和麦克风采集
        try await systemAudioCapture.start()
        try await microphoneCapture.start()
        
        // 2. 设置音频分离处理
        setupAudioSeparation()
    }
    
    private func setupAudioSeparation() {
        // 音频源分离算法
        systemAudioCapture.onAudioReceived = { [weak self] audioData in
            // 系统音频主要包含面试官声音
            self?.processInterviewerAudio(audioData)
        }
        
        microphoneCapture.onAudioReceived = { [weak self] audioData in
            // 麦克风音频主要包含候选人声音
            self?.processCandidateAudio(audioData)
        }
    }
    
    private func processInterviewerAudio(_ audioData: [Float]) {
        // 降噪处理
        let cleanedAudio = NoiseReduction.apply(to: audioData)
        
        // 声音增强
        let enhancedAudio = AudioEnhancement.amplify(cleanedAudio)
        
        DispatchQueue.main.async {
            self.interviewerAudio = enhancedAudio
            // 发送到语音识别
            self.sendToSpeechRecognition(enhancedAudio, source: .interviewer)
        }
    }
    
    private func processCandidateAudio(_ audioData: [Float]) {
        // 回声消除（重要：避免系统音频干扰）
        let echoFreeAudio = EchoCancellation.apply(to: audioData, reference: interviewerAudio)
        
        DispatchQueue.main.async {
            self.candidateAudio = echoFreeAudio
            self.sendToSpeechRecognition(echoFreeAudio, source: .candidate)
        }
    }
}
```

```swift
        // 捕获网页音频
        (function() {
            const audioContext = new (window.AudioContext || window.webkitAudioContext)();
            
            // 捕获页面音频
            navigator.mediaDevices.getDisplayMedia({ audio: true, video: false })
                .then(stream => {
                    const source = audioContext.createMediaStreamSource(stream);
                    const processor = audioContext.createScriptProcessor(4096, 1, 1);
                    
                    processor.onaudioprocess = function(e) {
                        const inputData = e.inputBuffer.getChannelData(0);
                        const audioArray = Array.from(inputData);
                        
                        // 发送音频数据到 Swift
                        window.webkit.messageHandlers.audioData.postMessage({
                            type: 'webAudio',
                            data: audioArray,
                            timestamp: Date.now()
                        });
                    };
                    
                    source.connect(processor);
                    processor.connect(audioContext.destination);
                })
                .catch(console.error);
        })();

        webView.configuration.userContentController.addUserScript(userScript)
        
        // 注册消息处理器
        webView.configuration.userContentController.add(self, name: "audioData")
    }
```

```swift
class IntelligentAudioRouter: ObservableObject {
    @Published var currentMode: AudioMode = .systemAudio
    @Published var audioQuality: Float = 0.0
    
    private var systemCapture = SystemAudioCapture()
    private var webCapture = WebAudioCapture()
    private var qualityAnalyzer = AudioQualityAnalyzer()
    
    func startCapture(mode: AudioMode) async throws {
        currentMode = mode
        
        switch mode {
        case .systemAudio:
            try await startSystemAudioMode()
        case .webAudio:
            try await startWebAudioMode()
        }
        
        // 启动质量监控
        startQualityMonitoring()
    }
    
    private func startSystemAudioMode() async throws {
        // 1. 启动系统音频采集
        try await systemCapture.start()
        
        // 2. 配置应用程序音频优先级
        configureAppAudioPriority([
            "com.tencent.meeting",      // 腾讯会议
            "com.bytedance.lark",       // 飞书
            "us.zoom.xos"               // Zoom
        ])
    }
    
    private func configureAppAudioPriority(_ bundleIDs: [String]) {
        for bundleID in bundleIDs {
            if let app = NSWorkspace.shared.runningApplications.first(where: { $0.bundleIdentifier == bundleID }) {
                // 提高特定应用的音频优先级
                setAudioPriority(for: app, priority: .high)
            }
        }
    }
    
    private func startQualityMonitoring() {
        Timer.scheduledTimer(withTimeInterval: 1.0, repeats: true) { [weak self] _ in
            self?.updateAudioQuality()
        }
    }
    
    private func updateAudioQuality() {
        let quality = qualityAnalyzer.analyzeCurrentAudio()
        
        DispatchQueue.main.async {
            self.audioQuality = quality
            
            // 如果质量太低，建议切换模式
            if quality < 0.3 {
                self.suggestModeSwitch()
            }
        }
    }
}
```
